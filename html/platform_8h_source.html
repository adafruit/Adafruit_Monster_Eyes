88 // PSRAM IS A TRAP FOR THIS WORKLOAD. With PSRAM configured, the default malloc
89 // sends large blocks to external RAM -- and the polar maps and iris texture are
90 // exactly that size. The render loop then does a random external read per
91 // pixel, which costs far more than the arithmetic around it.
92 //
93 // So eye data is allocated MALLOC_CAP_INTERNAL, and the texture budget is
94 // measured against internal RAM only. If internal RAM runs out the texture
95 // loader simply decimates further, which costs sharpness rather than speed.
96 static inline void *eyeMalloc(size_t n) {
97  void *p = heap_caps_malloc(n, MALLOC_CAP_INTERNAL | MALLOC_CAP_8BIT);
98  return p ? p : malloc(n); // Fall back rather than fail outright
99 }
