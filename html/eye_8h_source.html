142 // Byte order the render loop writes in. It costs nothing either way -- the
143 // swap happens once per texture at load and once per colour per frame -- so
144 // each backend asks for whatever its output path wants:
145 //
146 // RP2 burst native; 16-bit MSB-first frames emit the high byte first
147 // Adafruit TFT native; the driver swaps, and that branch measured faster
148 // DVI native; GFXcanvas16 is native uint16_t
149 // esp_lcd BIG-ENDIAN; the buffer goes to DMA verbatim
150 #if EYE_DISPLAY == EYE_DISPLAY_ESP_LCD
151 #define DISPLAY_BIG_ENDIAN 1
152 
157 #define OUT16(v) __builtin_bswap16((uint16_t)(v))
