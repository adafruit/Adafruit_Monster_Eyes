170 // Byte order the render loop writes in. It costs nothing either way -- the
171 // swap happens once per texture at load and once per colour per frame -- so
172 // each backend asks for whatever its output path wants:
173 //
174 // RP2 burst native; 16-bit MSB-first frames emit the high byte first
175 // Adafruit TFT native; the driver swaps, and that branch measured faster
176 // DVI native; GFXcanvas16 is native uint16_t
177 // esp_lcd BIG-ENDIAN; the buffer goes to DMA verbatim
178 #if EYE_DISPLAY == EYE_DISPLAY_ESP_LCD
179 #define DISPLAY_BIG_ENDIAN 1
180
